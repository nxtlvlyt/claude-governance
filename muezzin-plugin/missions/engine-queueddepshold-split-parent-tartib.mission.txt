// 2026-07-21): a SPLIT-status dependency (AUTORUN line "SPLIT missions/<dep>", never DONE --
// the autosplit replaces the parent with tartib children) satisfies a dependent mission ONLY
// if every one of its own children, per <dep-stem>._split-manifest.json, is itself DONE with
// a passing receipt. One level deep only -- a grandchild-split parent is a known, accepted
// limitation, not silently mis-handled (falls through to the normal unsatisfied hold, same
// as today, never a false-positive release).
function splitParentSatisfied(dep, autorunText, resultOkFn, readManifestFn) {
  const manifestRel = dep.replace(/\.mission\.txt$/i, '._split-manifest.json');
  let manifest;
  try { manifest = JSON.parse(readManifestFn(manifestRel)); } catch { return false; }
  const children = Array.isArray(manifest?.children) ? manifest.children : [];
  if (!children.length) return false;
  return children.every((c) => {
    const childFile = String(c?.file || '').replace(/\\/g, '/');
    if (!childFile) return false;
    const childDoneRe = new RegExp(`^DONE\\s+${childFile.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')}`, 'm');
    return childDoneRe.test(autorunText) && resultOkFn(childFile) === true;
  });
}
