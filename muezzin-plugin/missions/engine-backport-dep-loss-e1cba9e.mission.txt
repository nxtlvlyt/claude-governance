// ---- DECLARED-DEP VERBATIM SURVIVAL (backport-intake BUG 1, agy-muezzin commit e1cba9e,
// 2026-07-08 dep-loss receipted at atv-1 S1.S1). The parent mission DECLARED absolute
// context_dependencies verbatim in its step lines, but the plan is MODEL-AUTHORED --
// nothing mechanically forced the planner to copy the declared paths into the plan's
// context_dependencies, and a re-planned child INVENTED a substitute path and profiled
// against nothing. This is the MECHANICAL half: parse the dep declarations out of the
// mission text and re-inject any the planner dropped/re-authored, BEFORE validation, on
// every plan path (single architect + panel integrator, via runQueueLoop). The other half
// lives in mission_split.mjs emitSubMissions (already ported 2026-07-14), which serializes
// each child step's context_dependencies verbatim into the child mission text -- so the
// guarantee survives autosplit RECURSION (child text -> child plan -> grandchild text -> ...).
//
// Declaration convention (the parent's own shape, and what emitSubMissions emits):
//   ... [edit|command|verify] <target[, target]> ; context_dependencies: <dep[, dep]>
// A bare "context_dependencies: <dep[, dep]>" line (no same-line target) also counts --
// its deps fall back to every edit step. Tokens must be path-ish (no spaces/quotes/
// brackets, at least one '/', '\' or '.') so prose mentions and JSON schema fragments
// never parse as deps; "none" is the declared-empty marker and yields nothing.
const DEP_TOKEN_RE = /^[\w~][\w.:~\\/-]*$/;
const looksLikeDepPath = (s) => DEP_TOKEN_RE.test(s) && /[\\/.]/.test(s) && !/^none$/i.test(s);

// parseDeclaredDeps(mission) -> [{ deps: [verbatim...], targets: [verbatim...] }]
export function parseDeclaredDeps(mission) {
  const decls = [];
  for (const line of String(mission || '').split(/\r?\n/)) {
    const m = line.match(/^(.*?)\bcontext_dependencies\s*:\s*(.+)$/i);
    if (!m) continue;
    const deps = m[2].split(',').map((s) => s.trim()).filter(looksLikeDepPath);
    if (!deps.length) continue;
    const tm = m[1].match(/\[(?:edit|command|verify)\]\s*([^;[\]]*?)\s*;\s*$/i);
    const targets = tm ? tm[1].split(',').map((s) => s.trim()).filter(Boolean) : [];
    decls.push({ deps, targets });
  }
  return decls;
}

// injectDeclaredDeps(queue, mission, { research }) -- mutates queue in place; returns
// { injected, decls }. Pairing: a declaration carrying a same-line target lands on the
// plan step(s) that target that file; a declaration with no target (or whose target the
// planner renamed) falls back to every 'edit' step (deps are read-only inputs -- over-
// providing is bounded by executor dep-windowing; under-providing is the receipted bug).
// Class containment is MIRRORED, not bypassed: a dep validateMicroAction would reject
// for this class (absolute outside research, any '..') is skipped, never injected -- the
// backstop must not deadlock the repair loop on a mission-authoring error the mission
// lint already polices.
export function injectDeclaredDeps(queue, mission, { research = false } = {}) {
  const decls = parseDeclaredDeps(mission);
  const steps = Array.isArray(queue?.steps) ? queue.steps : [];
  let injected = 0;
  if (!decls.length || !steps.length) return { injected, decls: decls.length };
  const norm = (p) => String(p).toLowerCase().replace(/\\/g, '/');
  for (const { deps, targets } of decls) {
    const legal = deps.filter((d) => !escapesSandbox(d) || (research && String(d).indexOf('..') === -1));
    if (!legal.length) continue;
    const tset = new Set(targets.map(norm));
    let recipients = tset.size ? steps.filter((s) => (s?.target_files || []).some((t) => tset.has(norm(t)))) : [];
    if (!recipients.length) recipients = steps.filter((s) => s?.action_type === 'edit');
    if (!recipients.length) recipients = steps;
    for (const s of recipients) {
      if (!Array.isArray(s.context_dependencies)) s.context_dependencies = [];
      const have = new Set(s.context_dependencies.map(norm));
      for (const d of legal) {
        if (!have.has(norm(d))) { s.context_dependencies.push(d); have.add(norm(d)); injected++; }
      }
    }
  }
  return { injected, decls: decls.length };
}
