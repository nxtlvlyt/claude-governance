```
// CONDUCTOR-APPLY RECEIPT (gap-conductor-apply-lacks-daemon-receipt, filed 2026-07-18): a
// ninth-law in-turn conductor apply lands a mission (committed + witnessed) with a DONE
// AUTORUN line but writes no result.json -- only orchestrate()'s own fire-time write (line
// ~1132) ever produces one, so queuedDepsHold's dependency-satisfier check (line 882: DONE +
// resultOkFn(dep)===true) can never pass for a conductor-direct landing, and every downstream
// tartib sibling holds forever (live receipt: mt-feedback-capture.S1.S2 TARTIB-HELD ~30h
// behind S1.S1 for exactly this reason). This writes the SAME receipt shape at the SAME path
// convention resOk() reads (line ~1497: <mission stem>.mission.result.json), so the daemon's
// own gate accepts a conductor-applied mission identically to an orchestrate()-run one -- no
// daemon code path changes, no RESOLVED-comment workaround needed.
export function writeConductorApplyReceipt(missionFilePath, { commit, witness, note } = {}) {
  const receiptPath = String(missionFilePath).replace(/\.[^.]+$/, '') + '.result.json';
  const body = { ok: true, phase: 'conductor-apply', receipt: { commit, witness, note }, ts: new Date().toISOString() };
  writeFileSync(receiptPath, JSON.stringify(body, null, 2));
  return receiptPath;
}
```
