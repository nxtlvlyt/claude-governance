const REACHABILITY_NOTE_RE = /^REACHABILITY NOTE\s*\([^)\r\n]*\)\s*:.*?(?=\r?\n\s*\r?\n|\r?\n\d+\.\s|(?![\s\S]))/ims;
const UNLINKED_OK_LINE_RE = /^UNLINKED-OK:.*$/im;
const HTML_FILE_RE = /\.html$/i;
const ENTRY_HTML_RE = /(^|\/)(index|map)\.html$/i;

export function buildReachabilityDeclaration(files, parentMissionText) {
  const shipsPage = (files || []).some((f) => HTML_FILE_RE.test(f) && !ENTRY_HTML_RE.test(f));
  if (!shipsPage) return null;
  const t = String(parentMissionText || '');
  const noteMatch = t.match(REACHABILITY_NOTE_RE);
  if (noteMatch) return noteMatch[0].trim();
  const unlinkedMatch = t.match(UNLINKED_OK_LINE_RE);
  return unlinkedMatch ? unlinkedMatch[0].trim() : null;
}
