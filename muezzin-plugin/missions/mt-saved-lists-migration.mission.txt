-- ── saved_lists (M-SAVED-LIST: P1 competitor parity) ─────────────────────────
-- Every paid competitor (iOverlander, Park4Night, Campendium) ships a saved /
-- favorites primitive — most paywall it. We ship the FREE version: a logged-in
-- user can create up to MAX_LISTS lists, each holding up to MAX_ITEMS POIs or
-- reviews, with map fly-to from the bookmark icon in the POI popup and a
-- /saved.html management page.
--
-- Identity model mirrors the contribute worker contract (mission
-- photo-fix-contributor-leak): user_id is the canonical accounts.users.id (the
-- TRUSTED X-User-Id header the Pages proxy sets after /me succeeds), NOT the
-- email. Email round-trips fine for legacy /mine but the saved-list primitive
-- is new, so we key on the harder identifier directly. No body-supplied
-- contributor field is ever stored here; the dispatcher overwrites it with
-- the verified userId before INSERT.
--
-- target_kind ENUM('spot','review') matches the existing community_review
-- target ENUM('poi','spot'). Note we EXTEND with 'review' because users save
-- reviews/tips, not just spots — and we add 'poi' so users can save the
-- 30k-row legacy POIs too (community_spot is only the user-added ones).
CREATE TABLE IF NOT EXISTS saved_lists (
  id          INTEGER PRIMARY KEY AUTOINCREMENT,
  user_id     INTEGER NOT NULL,         -- accounts.users.id; set from X-User-Id (proxy) / verified token (worker)
  name        TEXT NOT NULL DEFAULT 'My spots',
  created_at  TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP
);
CREATE INDEX IF NOT EXISTS idx_saved_lists_user_id ON saved_lists (user_id);

CREATE TABLE IF NOT EXISTS saved_list_items (
  id           INTEGER PRIMARY KEY AUTOINCREMENT,
  list_id      INTEGER NOT NULL REFERENCES saved_lists(id),
  target_kind  TEXT NOT NULL CHECK (target_kind IN ('spot','review','poi')),
  target_id    INTEGER NOT NULL,
  note         TEXT,
  added_at     TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP
);
CREATE INDEX IF NOT EXISTS idx_saved_list_items_list_id ON saved_list_items (list_id);
-- Helps the "is POI X already in any of my lists?" bookmark-icon check (workers
-- query by list_id + target_kind + target_id when toggling).
CREATE INDEX IF NOT EXISTS idx_saved_list_items_target ON saved_list_items (target_kind, target_id);
