// SAME LOSS SHAPE as context_dependencies (commit e1cba9e): the parent mission DECLARED
// per-step validation_command gates verbatim (atv-1's anti-fabrication FETCH-RECEIPT
// Select-String gates), but the plan is MODEL-AUTHORED and an autosplit child is RE-PLANNED
// from its own text — so a gate absent from the child text is gone, and the child's planner
// RE-INVENTS a weaker witness (a bare Test-Path where the mission demanded the FETCH-RECEIPT
// content gate). This is the MECHANICAL half: parse the declared gates out of the mission
// text and re-impose any the planner re-authored, BEFORE validation, on every plan path
// (single architect + panel integrator, via runQueueLoop). The other half lives in
// mission_split.mjs emitSubMissions, which serializes each child step's validation_command
// verbatim as a following indented line — the parent's own convention — so the guarantee
// survives autosplit RECURSION (parent -> S1 -> S1.S1 -> ...).
//
// Declaration convention (the parent's own shape — atv-1 — and what emitSubMissions emits):
//   ... [edit|command|verify] <target[, target]> [; context_dependencies: <dep[, dep]>]
//       validation_command: <the complete witness command, verbatim to end of line>
// The command line binds to the NEAREST PRECEDING [action] step-target line; a blank line
// or a new numbered step line breaks the binding, and a binding is consumed by the first
// command line it receives — so prose paragraphs can never inherit a stale step. A same-line
// "... [action] <target> ; validation_command: <cmd>" also counts. Deps used a token filter
// (looksLikeDepPath) to reject prose; a command is arbitrary text, so here the anti-prose
// filter is STRUCTURAL: an unbound "validation_command:" line parses as NOTHING.
const ACTION_TARGETS_RE = /\[(edit|command|verify)\]\s*([^;\r\n]*)/i;

// parseDeclaredValidations(mission) -> [{ command: verbatim, targets: [verbatim...], action }]
export function parseDeclaredValidations(mission) {
  const decls = [];
  let bound = null;   // nearest preceding step-target line still eligible to receive a command
  for (const line of String(mission || '').split(/\r?\n/)) {
    if (!line.trim()) { bound = null; continue; }                        // blank line breaks binding
    const am = line.match(ACTION_TARGETS_RE);
    if (am) {
      const action = am[1].toLowerCase();
      const targets = am[2].split(',').map((s) => s.trim()).filter(Boolean);
      const sm = line.match(/\bvalidation_command\s*:\s*(.+)$/i);        // same-line form
      if (sm && targets.length && sm[1].trim()) { decls.push({ command: sm[1].trim(), targets, action }); bound = null; }
      else bound = targets.length ? { targets, action } : null;
      continue;
    }
    if (/^\s*\d+\.\s/.test(line)) { bound = null; continue; }            // new numbered step (its [action] line not seen yet)
    const vm = line.match(/^[ \t]*validation_command\s*:\s*(.+)$/i);     // next-line form (the atv-1 shape)
    if (vm && bound && vm[1].trim()) { decls.push({ command: vm[1].trim(), targets: bound.targets, action: bound.action }); bound = null; }
  }
  return decls;
}

// injectDeclaredValidations(queue, mission, { research, codeRepo }) — mutates queue in
// place; returns { injected, decls }. Pairing: a declaration lands on the plan step(s)
// matching BOTH its action_type and a declared target; if none, a target-only match still
// receives it (the planner changed the action but kept the deliverable). NO edit-step
// fallback — deliberately ASYMMETRIC with injectDeclaredDeps: deps are ADDITIVE read-only
// inputs (over-providing is bounded), but a validation_command is a scalar OVERWRITE, and
// spraying one gate over every step would destroy correct witnesses. A fully-renamed target
// keeps the planner's command (mission lint + the verdict panel still police it).
// Floor containment MIRRORED, not bypassed (the dep fix's escapesSandbox principle): a
// declared command that would trip a validateMicroAction floor the planner's own command
// does not is skipped, never injected — the backstop runs on every repair attempt and must
// not deadlock the repair loop on a mission-authoring error.
export function injectDeclaredValidations(queue, mission, { research = false, codeRepo = false } = {}) {
  const decls = parseDeclaredValidations(mission);
  const steps = Array.isArray(queue?.steps) ? queue.steps : [];
  let injected = 0;
  if (!decls.length || !steps.length) return { injected, decls: decls.length };
  const norm = (p) => String(p).toLowerCase().replace(/\\/g, '/');
  const vopts = { research, codeRepo };
  for (const { command, targets, action } of decls) {
    if (!command || !targets.length) continue;
    const tset = new Set(targets.map(norm));
    const onTarget = (s) => (s?.target_files || []).some((t) => tset.has(norm(t)));
    let recipients = steps.filter((s) => s?.action_type === action && onTarget(s));
    if (!recipients.length) recipients = steps.filter(onTarget);
    for (const s of recipients) {
      if (String(s.validation_command || '') === command) continue;      // idempotent — already verbatim
      const before = validateMicroAction(s, 0, vopts);
      const after = validateMicroAction({ ...s, validation_command: command }, 0, vopts);
      if (after.some((e) => !before.includes(e))) continue;              // declared gate trips a floor the planner's didn't -> skip (no deadlock)
      s.validation_command = command;
      injected++;
    }
  }
  return { injected, decls: decls.length };
}
